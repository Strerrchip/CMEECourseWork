#!/bin/bash
# Author: Leyu Chen
# Script: csvtospace.sh
# Description: Convert CSV (comma separated) to space-separated text file
# Usage: bash csvtospace.sh <input.csv>

if [ $# -ne 1 ]; then
    echo "Usage: bash csvtospace.sh <input.csv>"
    exit 1
fi

infile="$1"

if [ ! -f "$infile" ]; then
    echo "Error: file '$infile' not found!"
    exit 1
fi

outfile="${infile%.csv}_space.txt"

if [ -f "$outfile" ]; then
    echo "Error: output file '$outfile' already exists. Aborted."
    exit 1
fi

awk 'BEGIN{FS=","; OFS=" "} {for(i=1; i<=NF; i++) if ($i=="") $i="NA"; print $0}' "$infile" > "$outfile"

echo "Conversion done! Saved as '$outfile'."

# Desc: Convert all .tif images in the current directory to .png format
# Date: Oct 2025

# 检查是否安装了 convert 命令
if ! command -v convert &> /dev/null
then
    echo "Error: ImageMagick not installed. Please run: sudo apt install imagemagick"
    exit 1
fi

# 检查是否存在 .tif 文件
shopt -s nullglob
files=(*.tif)
if [ ${#files[@]} -eq 0 ]; then
    echo "No .tif files found in the current directory."
    exit 0
fi

# 转换所有 .tif 文件
for f in *.tif; do
    echo "Converting $f ..."
    convert "$f" "$(basename "$f" .tif).png"
