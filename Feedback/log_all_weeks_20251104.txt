  N <- matrix(NA, numyears, length(p0))  #initialize empty matrix

  N[1, ] <- p0

  for (pop in 1:length(p0)) { #loop through the populations

    for (yr in 2:numyears){ #for each pop, loop through the years

      N[yr, pop] <- N[yr-1, pop] * exp(r * (1 - N[yr - 1, pop] / K) + rnorm(1, 0, sigma)) # add one fluctuation from normal distribution
    
     }
  
  }

#!/usr/bin/env python3
# Author: Avery L Chen leyu.chen25@imperial.ac.uk
# Script: oaks_debugme_fixed.py
# Desc: Find oaks robustly; includes doctests and handles minor typos/punctuation
# Date: Oct 2025

import re
from pathlib import Path
import csv

def _levenshtein_one_or_less(a: str, b: str) -> bool:
    """Return True if edit distance between a and b is <= 1 (insert/delete/substitute).
    This lightweight check is enough for common typos like 'Quercuss' or 'Querucs'.
    """
    if a == b:
        return True
    la, lb = len(a), len(b)
    # length diff > 1 -> distance > 1
    if abs(la - lb) > 1:
        return False
    # same length: allow at most 1 substitution
    if la == lb:
        mism = sum(x != y for x, y in zip(a, b))
        return mism <= 1
    # length differs by 1: allow one insertion/deletion
    if la > lb:
        a, b = b, a  # make a the shorter
        la, lb = lb, la
    i = j = diff = 0
    while i < la and j < lb:
        if a[i] == b[j]:
            i += 1; j += 1
        else:
            diff += 1
            if diff > 1:
                return False
            j += 1  # skip one char in longer string
    # remaining chars in b also count
    diff += (lb - j)
    return diff <= 1

def is_an_oak(name: str) -> bool:
    """
    Return True if a species name belongs to genus Quercus (robust to case, punctuation, minor typos).

    Doctests:
    >>> is_an_oak('Quercus robur')
    True
    >>> is_an_oak('quercus cerris')
    True
    >>> is_an_oak('Fagus sylvatica')
    False
    >>> is_an_oak('Quercuss petraea')   # extra 's' -> treat as oak
    True
    >>> is_an_oak('Querucs suber')      # letters swapped -> treat as oak
    True
    >>> is_an_oak('Quercus. ilex')      # punctuation after genus
    True
    >>> is_an_oak('Quercu robur')       # distance=1 but missing 's' -> treat as oak
    True
    >>> is_an_oak('Que rcus robur')     # weird spacing -> clean then check
    True
    """
    if not name or not name.strip():
        return False
    first = name.strip().split()[0]
    first = re.sub(r'[^A-Za-z]', '', first).lower()  # remove punctuation, to lower
    target = "quercus"
    return _levenshtein_one_or_less(first, target)

def run_with_profile(cmd, prof_path):
    """Run a python command under cProfile, return wall time seconds."""
    t0 = time.perf_counter()
    subprocess.run(["python3", "-m", "cProfile", "-o", str(prof_path)] + cmd,
                   check=True)
    t1 = time.perf_counter()
    return t1 - t0

    print(f"\n=== Profile: {title} ({prof_path.name}) ===")

#!/usr/bin/env python3
# Author: Avery L Chen leyu.chen25@imperial.ac.uk
# Script: LV1.py
# Desc: Lotka–Volterra (consumer–resource) dynamics; save time-series and phase plot as PDFs
# Date: Oct 2025

"""
Integrate a simple consumer–resource (Lotka–Volterra) model using
an explicit Euler scheme (no SciPy required), and save two figures:

1) Time series of Resource (R) and Consumer (C) vs time
2) Phase portrait: C vs R

Both PDFs are saved in ../Results without displaying on screen.
"""

from pathlib import Path
import numpy as np

# Use a non-interactive backend so nothing pops up
import matplotlib
matplotlib.use("Agg")
import matplotlib.pyplot as plt


def lv_euler(R0=40.0, C0=9.0, a=0.6, b=0.025, c=0.8, d=0.02,  # parameters
             tmax=60.0, dt=0.01):
    """
    Euler integration for:
        dR/dt =  a*R - b*R*C
        dC/dt = -c*C + d*R*C
    Returns (t, R, C)
    """
    n = int(tmax / dt) + 1
    t = np.linspace(0, tmax, n)
    R = np.zeros(n)
    C = np.zeros(n)
    R[0], C[0] = R0, C0

    for i in range(n - 1):
        dR = a * R[i] - b * R[i] * C[i]
        dC = -c * C[i] + d * R[i] * C[i]
        R[i + 1] = max(R[i] + dt * dR, 0.0)  # keep non-negative
        C[i + 1] = max(C[i] + dt * dC, 0.0)

    return t, R, C


def main():
    # ---- run model
    t, R, C = lv_euler()

    # ---- output directory (Week2/Results)
    out_dir = Path(__file__).resolve().parent.parent / "Results"
    out_dir.mkdir(parents=True, exist_ok=True)

    # ---- Figure 1: time series
    fig1 = plt.figure()
    plt.plot(t, R, label="Resource (R)")
    plt.plot(t, C, label="Consumer (C)")
    plt.xlabel("Time")
    plt.ylabel("Density")
    plt.title("Lotka–Volterra: time series")
    plt.legend()
    fig1.tight_layout()
    fig1.savefig(out_dir / "LV_time.pdf")

    # ---- Figure 2: phase portrait (C vs R)
    fig2 = plt.figure()
    plt.plot(R, C)
    plt.xlabel("Resource density (R)")
    plt.ylabel("Consumer density (C)")
    plt.title("Consumer–Resource population dynamics")
    fig2.tight_layout()
    fig2.savefig(out_dir / "LV_phase.pdf")

    # plt.show()  # intentionally omitted


if __name__ == "__main__":

# Desc: Read two DNA sequences from a file, align the shorter to the longer by sliding,
#       and write the best alignment and score to an output text file.
# Date: Oct 2025

from pathlib import Path
import sys
import csv

def read_two_seqs_from_csv(infile):
    """Read two sequences from a CSV file with 2 rows (or any rows labelled seq1/seq2)."""
    s1, s2 = None, None
    with open(infile, newline="") as f:
        rdr = csv.reader(f)
        for row in rdr:
            if not row:
                continue
            # accept formats: ["seq1","ATCG..."] or ["ATCG...", "..."] etc.
            if len(row) == 2 and row[0].strip().lower().startswith("seq"):
                label, seq = row[0], row[1]
            else:
                # if no labels, take as sequence only
                seq = row[-1]
            seq = "".join(seq.split()).upper()   # remove whitespace, upper-case
            if s1 is None:
                s1 = seq
            elif s2 is None:
                s2 = seq
                break
    if s1 is None or s2 is None:
        raise ValueError("Could not read two sequences from input file.")
    return s1, s2

def calculate_score(longer, shorter, start_pos):
    """Return score for aligning shorter against longer starting at start_pos."""
    score = 0
    for i in range(len(shorter)):
        if start_pos + i >= len(longer):
            break
        if longer[start_pos + i] == shorter[i]:
            score += 1
    return score

def best_alignment(seq_a, seq_b):
    """Return tuple (best_offset, best_score, longer, shorter)."""
    # decide which one is longer
    if len(seq_a) >= len(seq_b):
        longer, shorter = seq_a, seq_b
    else:
        longer, shorter = seq_b, seq_a

    best_score = -1
    best_offset = 0
    # slide shorter along longer
    for start in range(len(longer) - len(shorter) + 1):
        sc = calculate_score(longer, shorter, start)
        if sc > best_score:
            best_score = sc
            best_offset = start
    return best_offset, best_score, longer, shorter

def make_alignment_string(longer, shorter, offset):
    """Return a human-readable alignment string with dots for mismatches."""
    pad = "." * offset
    aligned = pad + shorter
    # pad to the right to match length for printing
    if len(aligned) < len(longer):
        aligned = aligned + "." * (len(longer) - len(aligned))
    return aligned

def write_result(outfile, longer, aligned_shorter, score):
    outdir = Path(outfile).parent
    outdir.mkdir(parents=True, exist_ok=True)
    with open(outfile, "w") as g:
        g.write(f"Best alignment score: {score}\n")
        g.write(f"Longer : {longer}\n")
        g.write(f"Aligned: {aligned_shorter}\n")

def main():
    # default paths
    default_in = Path(__file__).resolve().parent.parent / "Data" / "sequences.csv"
    default_out = Path(__file__).resolve().parent.parent / "Results" / "best_alignment.txt"

    # allow optional CLI args: infile [outfile]

        infile = default_in
        outfile = default_out
    elif len(sys.argv) == 2:
        infile = Path(sys.argv[1])
        outfile = default_out
    else:
        infile = Path(sys.argv[1])
        outfile = Path(sys.argv[2])

    s1, s2 = read_two_seqs_from_csv(infile)
    offset, score, longer, shorter = best_alignment(s1, s2)
    aligned_shorter = make_alignment_string(longer, shorter, offset)
    write_result(outfile, longer, aligned_shorter, score)

    # also show to screen
    print(f"Score: {score}")
    print(f"Longer : {longer}")
    print(f"Aligned: {aligned_shorter}")
    print(f"\nWritten to: {outfile}")

if __name__ == "__main__":
    main()

#!/usr/bin/env python3
# Author: Avery L Chen leyu.chen25@imperial.ac.uk
# Script: LV2.py
# Desc: Lotka–Volterra with resource density dependence (logistic term); save PDFs, CLI params
# Date: Oct 2025

from pathlib import Path
import argparse
import numpy as np
import matplotlib
matplotlib.use("Agg")
import matplotlib.pyplot as plt

def lv2_euler(R0=40.0, C0=9.0, r=1.0, a=0.5, z=1.0, e=0.5, K=50.0,
              tmax=60.0, dt=0.01):
    """Euler integration for:
        dR/dt = r*R*(1 - R/K) - a*C*R
        dC/dt = -z*C + e*a*C*R
    Returns (t, R, C)
    """
    n = int(tmax / dt) + 1
    t = np.linspace(0, tmax, n)
    R = np.zeros(n); C = np.zeros(n)
    R[0], C[0] = R0, C0
    for i in range(n - 1):
        dR = r*R[i]*(1 - R[i]/K) - a*C[i]*R[i]
        dC = -z*C[i] + e*a*C[i]*R[i]
        R[i+1] = max(R[i] + dt*dR, 0.0)
        C[i+1] = max(C[i] + dt*dC, 0.0)
    return t, R, C

def save_figs(t, R, C, out_dir, tag="LV2"):
    out_dir = Path(out_dir); out_dir.mkdir(parents=True, exist_ok=True)
    # time series
    fig1 = plt.figure()
    plt.plot(t, R, label="Resource (R)")
    plt.plot(t, C, label="Consumer (C)")
    plt.xlabel("Time"); plt.ylabel("Density")
    plt.title(f"{tag}: time series")
    plt.legend(); fig1.tight_layout()
    fig1.savefig(out_dir / f"{tag}_time.pdf")
    plt.close(fig1)
    # phase portrait
    fig2 = plt.figure()
    plt.plot(R, C)
    plt.xlabel("Resource density (R)")
    plt.ylabel("Consumer density (C)")
    plt.title(f"{tag}: phase portrait")
    fig2.tight_layout()
    fig2.savefig(out_dir / f"{tag}_phase.pdf")
    plt.close(fig2)

def parse_args():
    p = argparse.ArgumentParser(

    )
    p.add_argument("r", type=float, nargs="?", default=1.0, help="resource growth rate r")

if __name__ == "__main__":
